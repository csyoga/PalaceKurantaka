\enginline{\small {Pawing the Leg like a Bull Pose}}\\
\kaninline{\small {ವೃಷಪಾದಕ್ಷೇಪ}}

\bigskip

%~ \begin{img}
%~ \makebox[\textwidth][c]{%

\begin{sans}
athottānāni āsanāni\!॥\\
uttānaṃ śayanaṃ kṛtvā aṅgulibhiḥ kandharāṃ baddhvā kūrparau militvā nitambena bhūmiṃ spṛṣṭvā ekaṃ pādaṃ prasārya ekaikena pādena savyadakṣiṇaṃ bhrāmayitvā vṛṣapādakṣepaṃ bhavati\!॥1\!॥
\end{sans}
\bigskip

%1.
\begin{multicols}{2}
\begin{eng}
\clrnormalheading{{\bf Now, the Supine Postures.}}  

\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum] [The yogi] should lie supine,  bind the neck with the fingers, join the elbows, touch the buttocks on the ground, extend one leg and rotate the other leg to the left and right. [This] is the \textbf{Pawing the Leg like a Bull [Pose]} ({\it vṛṣapādakṣepa}).
\end{enumerate}
\end{eng}

\begin{kan}
\clrnormalheading{{\bf ಈಗ, ಮಲಗಿಕೊಂಡು ಮಾಡುವ ಆಸನಗಳು}}  

\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum] ಯೋಗಿಯು ಮಲಗಿದ ಸ್ಥಿತಿಯಲ್ಲಿ ತನ್ನ ಬೆರಳುಗಳಿಂದ ಕುತ್ತಿಗೆಯನ್ನು ಹಿಡಿದು, ಮೊಣಕೈಗಳನ್ನು ಜೋಡಿಸಿರಬೇಕು, ಪೃಷ್ಠದ ಭಾಗವು ನೆಲಕ್ಕೆ ತಗಲುವಂತಿರಬೇಕು. ಒಂದು ಕಾಲನ್ನು ಚಾಚಿ, ಮತ್ತೊಂದು ಕಾಲನ್ನು ಎಡಕ್ಕೆ ಮತ್ತು ಬಲಕ್ಕೆ ತಿರುಗಿಸಬೇಕು. ಇದನ್ನು  \textbf{‘ವೃಷಪಾದಕ್ಷೇಪ’} (ಗೂಳಿಯು ತನ್ನ ಕಾಲನ್ನು ಆಡಿಸಿದಂತೆ) ಎನ್ನಲಾಗುತ್ತದೆ.
\end{enumerate}
\end{kan}
\end{multicols}

\enginline{\small {Iron-bar Pose}}\\

\end{tabular}

\begin{sans}
uttānaṃ śayanaṃ kṛtvā pādau militvā prasārya nitambaṃ bhūmau spṛṣṭvā hastābhyāṃ kandharāṃ\endnote{kandharāṃ] P, kandharaṃ M} baddhvā kumbhakaṃ kṛtvā tiṣṭhet\!। parighāsanaṃ bhavati\!॥2\!॥
\end{sans}
\bigskip

%2.
\begin{multicols}{2}
\begin{eng}
\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum] [The yogi] should lie supine, join and extend the legs, touch the buttocks on the ground, bind the neck with the hands, hold the breath and remain thus. [This] is the \textbf{Iron-bar Pose} ({\it parighāsana}).
\end{enumerate}
\end{eng}

\begin{kan}
\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum] ಮಲಗಿದ ಸ್ಥಿತಿಯಲ್ಲಿ ಕಾಲುಗಳನ್ನು ಜೋಡಿಸಿ ಚಾಚಬೇಕು. ಪೃಷ್ಠವು ನೆಲಕ್ಕೆ ತಗಲಿರಲಿ. ಕೈಗಳಿಂದ ಕುತ್ತಿಗೆಯನ್ನು ಹಿಡಿದು, ಉಸಿರನ್ನು ಬಿಗಿಹಿಡಿದು ಹಾಗೆಯೇ ಇರಬೇಕು. ಇದಕ್ಕೆ \textbf{‘ಪರಿಘಾಸನ’} (ಕಬ್ಬಿಣದ ಕಂಬದಂತಹ ಆಸನ) ಎನ್ನಲಾಗುತ್ತದೆ.
\end{enumerate}
\end{kan}
\end{multicols}

\begin{sans}
uttānaṃ śayanaṃ kūrparadvayaṃ nābhau sthāpayitvā ekaikaṃ hastaṃ prasārya nāsikāyāmaṅguṣṭhapradeśena dhṛtvā tallakṣyeṇa jaghana\-pradeśena dhṛtvā sthāpayet\!। paraśvāsanaṃ\endnote{paraśvāsanaṃ] M(ac) P, paraśvadhāsanaṃ M(pc)} bhavati\!॥3\!॥
\end{sans}
\bigskip

%3.
\begin{multicols}{2}
\begin{eng}
\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum]
[The yogi] should lie supine, place the elbows on the navel, extend one hand at a time, hold the nose by the thumb [of the other hand] with the gaze on it, while supporting [the position] with the region of the hips, and remain thus. [This] is the \textbf{Hatchet Pose} ({\it paraśvāsana}).
\end{enumerate}
\end{eng}

\begin{kan}
\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum] 
ಮಲಗಿದ ಸ್ಥಿತಿಯಲ್ಲಿ ಮೊಣಕೈಗಳನ್ನು ನಾಭಿಯ (ಹೊಕ್ಕುಳ) ಮೇಲೆ ಇರಿಸಬೇಕು. ಒಂದೊಂದೇ ಕೈಯನ್ನು ಚಾಚುತ್ತಾ, ಒಂದು ಕೈಯ ಹೆಬ್ಬೆರಳಿನಿಂದ ಮೂಗಿನ ತುದಿಯನ್ನು ಹಿಡಿದು ಅದರತ್ತಲೇ ನೋಡಬೇಕು. ಇಡೀ ಶರೀರವು ಸೊಂಟದ ಭಾಗದ ಮೇಲೆ ಸಮತೋಲನದಲ್ಲಿರಬೇಕು. ಇದನ್ನು  \textbf{‘ಪರಶ್ವಾಸನ’} (ಕೊಡಲಿಯ ಆಸನ) ಎನ್ನಲಾಗುತ್ತದೆ.
\end{enumerate}
\end{kan}
\end{multicols}

\begin{sans}
uttānaṃ śayanamekaikaṃ pādaṃ grīvāyāṃ vinyasya itarahastena pādāgraṃ gṛhītvā itarapādahastau lambīkṛtya tiṣṭhet\!। anantāsanaṃ bhavati\!॥4\!॥
\end{sans}
\bigskip

%4.
\begin{multicols}{2}
\begin{eng}
\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum] [The yogi] should lie supine, fix one foot on [the back of] the neck, grasp the toes of [that] foot with the other hand, and stretch out the other foot and hand. He should remain thus [and then repeat it with the other foot on the neck]. [This] is \textbf{Ananta’s Pose} ({\it anantāsana}).
\end{enumerate}
\end{eng}

\begin{kan}
\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum]  ಮಲಗಿದ ಸ್ಥಿತಿಯಲ್ಲಿ ಒಂದು ಪಾದವನ್ನು ಕುತ್ತಿಗೆಯ ಹಿಂಭಾಗದಲ್ಲಿ ಇರಿಸಿ, ಅದೇ ಪಾದದ ಬೆರಳುಗಳನ್ನು ಕೈಯಿಂದ ಹಿಡಿಯಬೇಕು. ಉಳಿದ ಒಂದು ಕಾಲು ಮತ್ತು ಕೈಯನ್ನು ನೇರವಾಗಿ ಚಾಚಬೇಕು. ಇದೇ ರೀತಿ ಮತ್ತೊಂದು ಕಾಲಿನಿಂದಲೂ ಮಾಡಬೇಕು. ಇದು \textbf{‘ಅನಂತಾಸನ’}.
\end{enumerate}
\end{kan}
\end{multicols}

\begin{sans}
uttānaṃ śayīta\endnote{śayīta~] \textit{emend.}, śayātaṃ M.}\!। ekaikaṃ pādaṃ grīvāyāṃ kurvannitarahastaṃ karṇamūle sthāpayet\!। tasyaiva hastasya kūrparaṃ bhūmau nidhāya itarahastapādau  saralīkṛtya tiṣṭhet\!। aṅkuśāsanaṃ bhavati\!॥5\!॥
\end{sans}
\bigskip

%5.
\begin{multicols}{2}
\begin{eng}
\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum]
[The yogi] should lie supine. Putting one foot on the [back of the] neck, he should place the other hand on the root of the ear. Having placed the elbow of this arm on the ground, while making the other hand and leg straight, he should remain thus [and then repeat it with the other foot on the neck]. [This] is the \textbf{Goad Pose} ({\it aṅkuśāsana}).
\end{enumerate}
\end{eng}

\begin{kan}
\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum]
ಮಲಗಿದ ಸ್ಥಿತಿಯಲ್ಲಿ ಒಂದು ಪಾದವನ್ನು ಕುತ್ತಿಗೆಯ ಹಿಂದೆ ಇರಿಸಿ, ಒಂದು ಕೈಯನ್ನು ಕಿವಿಯ ಬುಡದ ಹತ್ತಿರ ಇಡಬೇಕು. ಆ ತೋಳಿನ ಮೊಣಕೈ ನೆಲಕ್ಕೆ ತಗಲಿರಲಿ. ಉಳಿದ ಒಂದು ಕೈ ಮತ್ತು ಕಾಲು ನೇರವಾಗಿರಬೇಕು. ಇದೇ ರೀತಿ ಎರಡೂ ಕಡೆ ಅಭ್ಯಾಸ ಮಾಡಬೇಕು. ಇದನ್ನು \textbf{‘ಅಂಕುಶಾಸನ’} ಎನ್ನಲಾಗುತ್ತದೆ.
\end{enumerate}
\end{kan}
\end{multicols}

śavavaccharīraṃ\endnote{śavavac°~] P, śavava M.} saṃsthāpya jānunī saṃmīlya nābhāvānīya hastābhyāṃ kandharāṃ baddhvā bhrāmayet\!। śvottānaṃ bhavati\!॥6\!॥

\end{enumerate}
\end{eng}

\begin{kan}
\begin{enumerate}[leftmargin=*,labelindent=0pt]
\item[\asananum]
ಶವದಂತೆ ನೇರವಾಗಿ ಮಲಗಿ, ಎರಡು ಮೊಣಕಾಲುಗಳನ್ನು ಜೋಡಿಸಿ ನಾಭಿಯ ಹತ್ತಿರಕ್ಕೆ ತರಬೇಕು. ಕೈಗಳಿಂದ ಕುತ್ತಿಗೆಯನ್ನು ತಬ್ಬಿಕೊಂಡು ಕಾಲುಗಳನ್ನು ತಿರುಗಿಸಬೇಕು. ಇದನ್ನು \textbf{‘ಶ್ವೋತ್ತಾನಾಸನ’} (ಮೇಲ್ಮುಖವಾದ ನಾಯಿಯ ಭಂಗಿ) ಎಂದು ಕರೆಯಲಾಗುತ್ತದೆ.
\end{enumerate}
\end{kan}
\end{multicols}
