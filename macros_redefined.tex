
\RenewDocumentCommand{\Section}{s O{} m}{%
  % #1 = * (True if starred, False if unstarred)
  % #2 = Optional short title for TOC
  % #3 = Mandatory section heading text
  %\par\vspace{1.5em}
  \noindent
  \IfBooleanTF{#1}{%
    % ----------------------------------------------------
    % STARRED VARIANT: Unnumbered + TOC Entry (No Hanging Indent Needed)
    % ----------------------------------------------------
    \iflanguage{kannada}{%
      {\fontsize{18pt}{21.6pt}\selectfont\color{deepmaroon}{#3}}%
      \addcontentsline{tocka}{section}{%
        \kantoc{\ifstrempty{#2}{#3}{#2}}%
      }%
    }{%
      {\cormorant\fontsize{18pt}{21.6pt}\selectfont\color{deepmaroon}{#3}}%
      \addcontentsline{toc}{section}{%
        \ifstrempty{#2}{#3}{#2}%
      }%
    }%
  }{%
    % ----------------------------------------------------
    % UNSTARRED VARIANT: Numbered + Hanging Indentation + TOC Entry
    % ----------------------------------------------------
    \refstepcounter{section}%
    \iflanguage{kannada}{%
      % 1. Measure the exact width of "\thesection. " (or "\thesection\quad ")
      \sbox{\secnumbox}{\fontsize{18pt}{21.6pt}\selectfont\color{deepmaroon}{\thesection.\quad}}%
      % 2. Set hanging indent for wrapped lines equal to the measured width
      \hangindent=\wd\secnumbox
      \hangafter=1
      % 3. Typeset heading
      {\fontsize{18pt}{21.6pt}\selectfont\color{deepmaroon}{\usebox{\secnumbox}#3}}%
      % 4. Add to .tocka
      \addcontentsline{tocka}{section}{%
        \protect\numberline{\thesection}%
        \kantoc{\ifstrempty{#2}{#3}{#2}}%
      }%
    }{%
      % 1. Measure the exact width of English section number
      \sbox{\secnumbox}{\cormorant\fontsize{18pt}{21.6pt}\selectfont\color{deepmaroon}{\thesection.\quad}}%
      % 2. Set hanging indent
      \hangindent=\wd\secnumbox
      \hangafter=1
      % 3. Typeset heading
      {\cormorant\fontsize{18pt}{21.6pt}\selectfont\color{deepmaroon}{\usebox{\secnumbox}#3}}%
      % 4. Add to standard .toc
      \addcontentsline{toc}{section}{%
        \protect\numberline{\thesection}%
        \ifstrempty{#2}{#3}{#2}%
      }%
    }%
  }%
  %\par\vspace{1em}
  \noindent\ignorespaces%
}

\RenewDocumentCommand{\SSubsection}{s O{} m}{%
  % #1 = * (True if starred, False if unstarred)
  % #2 = Optional short title for TOC (defaults to empty)
  % #3 = Mandatory subsubsection heading text
  %\par\vspace{1.0em}
  \noindent
  \IfBooleanTF{#1}{%
    % ----------------------------------------------------
    % STARRED VARIANT: Unnumbered + TOC Entry
    % ----------------------------------------------------
    \iflanguage{kannada}{%
      {\color{deepmaroon}{#3}}%
      \addcontentsline{tocka}{subsubsection}{%
        \kantoc{\ifstrempty{#2}{#3}{#2}}%
      }%
    }{%
      {{\cormorant\selectfont\color{deepmaroon}{#3}}}%
      \addcontentsline{toc}{subsubsection}{%
        \ifstrempty{#2}{#3}{#2}%
      }%
    }%
  }{%
    % ----------------------------------------------------
    % UNSTARRED VARIANT: Numbered + Hanging Indent + TOC Entry
    % ----------------------------------------------------
    \refstepcounter{subsubsection}% Increment subsubsection counter
    \iflanguage{kannada}{%
      % 1. Measure exact width of "\thesubsubsection.\quad"
      \sbox{\ssubsecnumbox}{\thesubsubsection.\quad}%
      % 2. Set hanging indent equal to the measured width
      \hangindent=\wd\ssubsecnumbox
      \hangafter=1
      % 3. Typeset heading
      {\usebox{\ssubsecnumbox}\color{deepmaroon}{#3}}%
      % 4. Add to .tocka File
      \addcontentsline{tocka}{subsubsection}{%
        \protect\numberline{\thesubsubsection}%
        \kantoc{\ifstrempty{#2}{#3}{#2}}%
      }%
    }{%
      % 1. Measure exact width of English subsubsection number
      \sbox{\ssubsecnumbox}{{\cormorant\selectfont\thesubsubsection.\quad}}%
      % 2. Set hanging indent
      \hangindent=\wd\ssubsecnumbox
      \hangafter=1
      % 3. Typeset heading
      {{\cormorant\selectfont\usebox{\ssubsecnumbox}\color{deepmaroon}{#3}}}%
      % 4. Add to standard .toc File
      \addcontentsline{toc}{subsubsection}{%
        \protect\numberline{\thesubsubsection}%
        \ifstrempty{#2}{#3}{#2}%
      }%
    }%
  }%
  %\par\vspace{0.6em}
  \noindent\ignorespaces%
}
